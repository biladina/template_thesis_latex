\essection{Judul Penelitian}
[Tuliskan judul yang spesifik, ringkas, dan merepresentasikan solusi atau metode sains
data/kecerdasan buatan yang akan dikembangkan. Maksimal 15-20 kata.]

\essection{Latar Belakang dan Rumusan Masalah}
\begin{espoints}
  \espoint{Konteks Masalah}{Jelaskan secara singkat fenomena, urgensi, atau masalah dunia
    nyata/teoritis yang memicu penelitian ini. Mengapa masalah ini penting untuk
    diselesaikan?}
  \espoint{Rumusan Masalah}{Sebutkan 1-3 pertanyaan penelitian utama yang akan dijawab
    melalui tesis ini.}
\end{espoints}

\essection{State of the Art (SOTA) dan Gap Penelitian}
\begin{espoints}
  \espoint{Kajian SOTA}{Ringkas 3-5 penelitian atau literatur paling relevan dan terbaru
    (diutamakan 3 tahun terakhir) yang telah mencoba menyelesaikan masalah serupa.
    Sebutkan keunggulan dari masing-masing penelitian tersebut.}
  \espoint{Gap Penelitian}{Jelaskan batasan atau kelemahan dari SOTA tersebut yang belum
    terselesaikan. Tunjukkan di titik mana penelitian tesis ini akan masuk untuk mengisi
    kekosongan (gap) tersebut.}
\end{espoints}

\essection{Tujuan dan Kontribusi Penelitian}
\begin{espoints}
  \espoint{Tujuan}{Apa yang ingin dicapai secara spesifik? (Misal: Mengembangkan algoritma
    X untuk meningkatkan akurasi Y pada kasus Z).}
  \espoint{Kebaruan (Novelty) / Kontribusi Utama}{Apa nilai tambah dari penelitian ini
    terhadap bidang sains data dan kecerdasan buatan? (Bisa berupa kebaruan arsitektur,
    optimasi model, dataset baru, atau implementasi praktis yang belum pernah ada).}
\end{espoints}

\essection{Ringkasan Metodologi}
\begin{esinstruction}
[Jelaskan secara garis besar pendekatan teknis yang akan digunakan. Poin-poin ini harus
mencakup hal teknis berikut:]
\end{esinstruction}
\begin{espoints}
  \espoint{Pengumpulan \& Sumber Data}{Dari mana data diperoleh? Berapa ukurannya? Apa
    formatnya?}
  \espoint{Pendekatan Model/Algoritma}{Algoritma Machine Learning/Deep Learning, model
    matematis, atau framework AI apa yang akan digunakan atau dimodifikasi?}
  \espoint{Skenario Evaluasi}{Metrik apa yang akan digunakan untuk mengukur keberhasilan
    model? (Misal: RMSE, F1-Score, Inference Time, dll).}
\end{espoints}
